cmake_minimum_required(VERSION 3.25)
project(MLTAStudio VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MLTA_STUDIO_ENABLE_QT "Build Qt 6 desktop interface when available" ON)
option(MLTA_STUDIO_BUILD_TESTS "Build MLTA engine and API tests" ON)
add_library(mlta_engine engine/src/engine.cpp)
target_include_directories(mlta_engine PUBLIC engine/include)
target_compile_features(mlta_engine PUBLIC cxx_std_23)
add_library(mlta_api api/src/router.cpp)
target_include_directories(mlta_api PUBLIC api/include)
target_link_libraries(mlta_api PUBLIC mlta_engine)
target_compile_features(mlta_api PUBLIC cxx_std_23)
add_library(mlta_catalog catalog/src/catalog.cpp)
target_include_directories(mlta_catalog PUBLIC catalog/include)
target_compile_features(mlta_catalog PUBLIC cxx_std_23)
add_library(mlta_generator generator/src/generator.cpp)
target_include_directories(mlta_generator PUBLIC generator/include)
target_link_libraries(mlta_generator PUBLIC mlta_engine)
target_compile_features(mlta_generator PUBLIC cxx_std_23)
foreach(target mlta_engine mlta_api mlta_catalog mlta_generator)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endforeach()
if(MLTA_STUDIO_ENABLE_QT)
  find_package(Qt6 6.4 COMPONENTS Widgets QUIET)
endif()
if(Qt6_FOUND)
  qt_standard_project_setup()
  qt_add_executable(mlta_studio src/main.cpp src/main_window.cpp include/mlta_studio/main_window.hpp)
  target_include_directories(mlta_studio PRIVATE include)
  target_link_libraries(mlta_studio PRIVATE Qt6::Widgets mlta_engine mlta_api mlta_catalog mlta_generator)
  target_compile_definitions(mlta_studio PRIVATE MLTA_STUDIO_WITH_QT=1)
else()
  add_executable(mlta_studio src/main.cpp)
  target_link_libraries(mlta_studio PRIVATE mlta_engine mlta_api mlta_catalog mlta_generator)
  message(STATUS "Qt 6 Widgets not found: building the MLTA Studio CLI bootstrap.")
endif()
if(MSVC)
  target_compile_options(mlta_studio PRIVATE /W4 /permissive-)
else()
  target_compile_options(mlta_studio PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(MLTA_STUDIO_BUILD_TESTS)
  enable_testing()
  add_executable(mlta_engine_tests tests/test_engine.cpp)
  target_link_libraries(mlta_engine_tests PRIVATE mlta_engine mlta_api mlta_generator)
  add_test(NAME mlta_engine_tests COMMAND mlta_engine_tests)
endif()
